{{/* Sequential question picker — walks the "questions" list top-to-bottom via a
     persisted counter ("questions_idx") instead of random, so nothing repeats
     until the whole list has been shown, then it wraps back to the top. dbIncr
     is atomic, so simultaneous uses still advance cleanly. 2 DB ops: read the
     list + bump the counter. */}}
{{- $q := (dbGet 0 "questions").Value -}}
{{- if $q -}}
{{- $c := toInt (dbIncr 0 "questions_idx" 1) -}}
{{- $i := toInt (mod (sub $c 1) (len $q)) -}}
{{- index $q $i -}}
{{- else -}}
No questions loaded yet.
{{- end -}}
